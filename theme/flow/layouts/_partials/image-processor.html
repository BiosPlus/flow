Full path attempted: '%s'" $src $imageAssetPath -}} {{- return -}} {{- end -}} {{- $isGif := eq $image.MediaType.SubType "gif" -}} {{- $hasCaption := or $caption $class -}} {{- if $hasCaption -}} {{- end -}} {{- if $isGif -}} {{- else -}} {{- $quality := .quality | default "75" -}} {{- /* Need to process AVIF and WebP */ -}} {{- $avifProcessOpt := printf "avif q%s" $quality -}} {{- $webpProcessOpt := printf "webp q%s" $quality -}} {{- $avifImage := $image.Process $avifProcessOpt -}} {{- $webpImage := $image.Process $webpProcessOpt -}} {{- end -}} {{- if $hasCaption -}} {{- with $caption -}} {{- end -}} {{- end -}}
