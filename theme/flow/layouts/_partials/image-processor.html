{{- $src := .src -}} {{- if not $src -}} {{- errorf "image-processor: 'src' parameter is required." -}} {{- return -}} {{- end -}} {{- $alt := .alt | default $src -}} {{- $caption := .caption -}} {{- $class := .class -}} {{- $loading := .loading | default "lazy" -}} {{- $imgWidth := .width -}} {{- $imgHeight := .height -}} {{- $quality := .quality | default "75" -}} {{- $cleanSrc := strings.TrimPrefix "/" $src -}} {{- $cleanSrc = strings.TrimPrefix "images/" $cleanSrc -}} {{- $imageAssetPath := printf "images/%s" $cleanSrc -}} {{- $resource := resources.Get $imageAssetPath -}} {{- if not $resource -}} {{- errorf "image-processor: Image '%s' not found in 'assets/images/'. Full path attempted: '%s'" $src $imageAssetPath -}} {{- return -}} {{- end -}} {{- $basePath := replaceRE `\.[^.]+$` "" $imageAssetPath -}} {{- $ext := lower (replaceRE `^.*\.` "" $cleanSrc) -}} {{- $isGif := or (eq $ext "gif") (eq $resource.MediaType.SubType "gif") -}} {{- $isWebm := or (eq $ext "webm") (eq $resource.MediaType.SubType "webm") -}} {{- $hasCaption := or $caption $class -}} {{- if $hasCaption -}} {{- end -}} {{- if or $isGif $isWebm -}} {{- /* Animated GIF / WebM handling */ -}} {{- $webmResource := resources.Get (printf "%s.webm" $basePath) -}} {{- $gifResource := resources.Get (printf "%s.gif" $basePath) -}} {{- $fallbackResource := or $gifResource $resource -}} {{- $w := or $imgWidth (and $fallbackResource $fallbackResource.Width) -}} {{- $h := or $imgHeight (and $fallbackResource $fallbackResource.Height) -}} {{- if $webmResource -}} {{- else -}} {{- end -}} {{- else -}} {{- /* Static image handling: JXL -> WebP failover -> Original fallback */ -}} {{- $jxlResource := resources.Get (printf "%s.jxl" $basePath) -}} {{- $w := or $imgWidth (and $resource.Width $resource.Width) -}} {{- $h := or $imgHeight (and $resource.Height $resource.Height) -}} {{- if eq $ext "jxl" -}} {{- $webpResource := resources.Get (printf "%s.webp" $basePath) -}} {{- with $webpResource -}} {{- end -}} {{- else -}} {{- $webpImage := $resource.Process (printf "webp q%s" $quality) -}} {{- with $jxlResource -}} {{- end -}} {{- with $webpImage -}} {{- end -}} {{- end -}} {{- end -}} {{- if $hasCaption -}} {{- with $caption -}} {{- end -}} {{- end -}}
